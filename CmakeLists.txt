cmake_minimum_required(VERSION 3.22)
project(photo_synth_plugin VERSION 1.0.0)

set(JUCE_PATH "./JUCE")
add_subdirectory(${JUCE_PATH} JUCE)

juce_add_plugin(photo_synth_plugin
    COMPANY_NAME "PhotoSynth"
    IS_SYNTH TRUE
    NEEDS_MIDI TRUE
    PLUGIN_MANUFACTURER_CODE Phsy
    PLUGIN_CODE Psyn
    FORMATS VST3 AU
    PRODUCT_NAME "Photo Synth")

juce_generate_juce_header(photo_synth_plugin)

target_sources(photo_synth_plugin
    PRIVATE
        PluginProcessor.cpp
        PluginProcessor.h
        PluginEditor.cpp
        PluginEditor.h
        SynthEngine.cpp
        SynthEngine.h
        SynthVoice.cpp
        SynthVoice.h
        ImageAnalyzer.cpp
        ImageAnalyzer.h
        MorphEngine.cpp
        MorphEngine.h
        PhysicalModeling.cpp
        PhysicalModeling.h
        EffectsChain.cpp
        EffectsChain.h
        CommonTypes.h)

target_link_libraries(photo_synth_plugin
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

set_target_properties(photo_synth_plugin PROPERTIES
    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY ""
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO"
)
